cmake_minimum_required(VERSION 3.28)
project(obs-parameter-animator VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_PLUGIN "Build native OBS module and Qt panel" ON)
include(CTest)
if(BUILD_TESTING)
  add_executable(animation-tests tests/animation_tests.cpp)
  target_include_directories(animation-tests PRIVATE src)
  add_test(NAME animation-engine COMMAND animation-tests)
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME original-shader-preserved COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/verify_shader.py)
    add_test(NAME shader-compiles COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compile_shader.py)
  endif()
endif()
if(BUILD_PLUGIN)
  find_package(libobs CONFIG REQUIRED)
  find_package(obs-frontend-api CONFIG REQUIRED)
  find_package(Qt6 CONFIG REQUIRED COMPONENTS Core Gui Widgets)
  add_library(obs-parameter-animator MODULE src/plugin-main.cpp src/filter.cpp src/ui.cpp src/ws_vendor.cpp
              src/face_tracker.cpp)
  target_include_directories(obs-parameter-animator PRIVATE include src)
  if(TARGET OBS::obs-frontend-api)
    set(OPA_FRONTEND_TARGET OBS::obs-frontend-api)
  elseif(TARGET OBS::frontend-api)
    set(OPA_FRONTEND_TARGET OBS::frontend-api)
  else()
    message(FATAL_ERROR "OBS frontend API target not exported by this SDK")
  endif()
  target_link_libraries(obs-parameter-animator PRIVATE OBS::libobs ${OPA_FRONTEND_TARGET} Qt6::Core Qt6::Gui Qt6::Widgets)
  target_compile_definitions(obs-parameter-animator PRIVATE PLUGIN_VERSION="${PROJECT_VERSION}")
  # Optional face tracking: YuNet (FaceDetectorYN) via OpenCV. When OpenCV is not found the plugin
  # still builds; the tracker is compiled as an inert stub (OPA_FACE_TRACKING undefined).
  option(OPA_FACE_TRACKING "Build YuNet face tracking (requires OpenCV)" ON)
  if(OPA_FACE_TRACKING)
    # OpenCVConfig.cmake only maps MSVC versions up to 19[34]x (vc17). With a newer toolset it
    # leaves OpenCV_RUNTIME empty and reports "no compatible binaries". Pick the best prebuilt
    # runtime folder that actually exists next to the given OpenCV_DIR.
    if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT DEFINED OpenCV_RUNTIME AND DEFINED OpenCV_DIR)
      foreach(opa_rt vc17 vc16 vc15)
        if(EXISTS "${OpenCV_DIR}/x64/${opa_rt}/lib/OpenCVConfig.cmake")
          set(OpenCV_ARCH x64)
          set(OpenCV_RUNTIME ${opa_rt})
          break()
        endif()
      endforeach()
    endif()
    find_package(OpenCV QUIET COMPONENTS core imgproc objdetect dnn calib3d)
    if(OpenCV_FOUND)
      target_compile_definitions(obs-parameter-animator PRIVATE OPA_FACE_TRACKING=1)
      target_include_directories(obs-parameter-animator PRIVATE ${OpenCV_INCLUDE_DIRS})
      # The debug OpenCV import lib uses cv::debug_build_guard (a different ABI), so linking it
      # into a release plugin leaves cv::cvtColor-style symbols unresolved. Always take the
      # release import location when the world target provides one.
      set(OPA_OCV_LIB "")
      if(TARGET opencv_world)
        get_target_property(OPA_OCV_LIB opencv_world IMPORTED_IMPLIB_RELEASE)
        if(NOT OPA_OCV_LIB)
          get_target_property(OPA_OCV_LIB opencv_world IMPORTED_IMPLIB)
        endif()
        if(NOT OPA_OCV_LIB)
          get_target_property(OPA_OCV_LIB opencv_world IMPORTED_LOCATION_RELEASE)
        endif()
      endif()
      if(OPA_OCV_LIB)
        target_link_libraries(obs-parameter-animator PRIVATE ${OPA_OCV_LIB})
      elseif(TARGET opencv_world)
        target_link_libraries(obs-parameter-animator PRIVATE opencv_world)
      else()
        target_link_libraries(obs-parameter-animator PRIVATE ${OpenCV_LIBS})
      endif()
      message(STATUS "OPA: face tracking enabled with OpenCV ${OpenCV_VERSION} (${OPA_OCV_LIB})")
      # Synthetic head-pose check (needs OpenCV but no camera).
      if(BUILD_TESTING)
        add_executable(pose-tests tests/pose_tests.cpp)
        target_include_directories(pose-tests PRIVATE src ${OpenCV_INCLUDE_DIRS})
        target_compile_definitions(pose-tests PRIVATE OPA_FACE_TRACKING=1)
        if(OPA_OCV_LIB)
          target_link_libraries(pose-tests PRIVATE ${OPA_OCV_LIB})
        else()
          target_link_libraries(pose-tests PRIVATE ${OpenCV_LIBS})
        endif()
        add_test(NAME head-pose COMMAND pose-tests)
        # Dense-landmark model smoke test (MediaPipe Face Mesh ONNX + canonical 3D model).
        add_executable(mesh-tests tests/mesh_tests.cpp)
        target_include_directories(mesh-tests PRIVATE src ${OpenCV_INCLUDE_DIRS})
        target_compile_definitions(mesh-tests PRIVATE OPA_FACE_TRACKING=1)
        if(OPA_OCV_LIB)
          target_link_libraries(mesh-tests PRIVATE ${OPA_OCV_LIB})
        else()
          target_link_libraries(mesh-tests PRIVATE ${OpenCV_LIBS})
        endif()
        add_test(NAME face-mesh
                 COMMAND mesh-tests "${CMAKE_CURRENT_SOURCE_DIR}/data/face_landmarks.onnx"
                         "${CMAKE_CURRENT_SOURCE_DIR}/data/canonical_face_model.obj")
        # The test executable needs the OpenCV runtime next to it (the plugin only ships it on
        # install, so the test would otherwise fail to start with a missing-DLL dialog).
        if(TARGET opencv_world)
          get_target_property(_opa_ocv_dll opencv_world IMPORTED_LOCATION_RELEASE)
          if(NOT _opa_ocv_dll)
            get_target_property(_opa_ocv_dll opencv_world IMPORTED_LOCATION)
          endif()
          if(_opa_ocv_dll)
            get_filename_component(_opa_ocv_name "${_opa_ocv_dll}" NAME_WE)
            get_filename_component(_opa_ocv_dir "${_opa_ocv_dll}" DIRECTORY)
            get_filename_component(_opa_ocv_bin "${_opa_ocv_dir}/../bin" ABSOLUTE)
            if(EXISTS "${_opa_ocv_bin}/${_opa_ocv_name}.dll")
              set(_opa_ocv_dllpath "${_opa_ocv_bin}/${_opa_ocv_name}.dll")
            elseif(EXISTS "${_opa_ocv_dir}/${_opa_ocv_name}.dll")
              set(_opa_ocv_dllpath "${_opa_ocv_dir}/${_opa_ocv_name}.dll")
            endif()
            if(_opa_ocv_dllpath)
              add_custom_command(TARGET pose-tests POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_opa_ocv_dllpath}"
                        "$<TARGET_FILE_DIR:pose-tests>")
              add_custom_command(TARGET mesh-tests POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${_opa_ocv_dllpath}"
                        "$<TARGET_FILE_DIR:mesh-tests>")
            endif()
          endif()
        endif()
      endif()
    else()
      message(WARNING "OPA: OpenCV not found; building without face tracking")
    endif()
  endif()
  # Visual editor preview: renders src/zone_canvas.hpp to PNG for review without OBS.
  if(BUILD_TESTING)
    add_executable(canvas-preview tests/canvas_preview.cpp)
    target_include_directories(canvas-preview PRIVATE src)
    target_link_libraries(canvas-preview PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
    if(WIN32)
      add_test(NAME zone-canvas-interaction
               COMMAND canvas-preview --selftest --qt-plugins "$<TARGET_FILE_DIR:Qt6::Core>/../plugins")
      # Qt builds shipped with OBS (obs-deps) only bundle the minimal/direct2d/windows
      # platform plugins, while a standalone Qt install also has "offscreen". Qt accepts a
      # fallback list, so prefer offscreen and fall back to minimal where it is absent.
      set_tests_properties(zone-canvas-interaction PROPERTIES
        ENVIRONMENT "QT_QPA_PLATFORM=offscreen\\;minimal;PATH=$<TARGET_FILE_DIR:Qt6::Core>;$ENV{PATH}")
    endif()
  endif()
  set_target_properties(obs-parameter-animator PROPERTIES PREFIX "")
  if(MSVC)
    target_compile_options(obs-parameter-animator PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(obs-parameter-animator PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(WIN32)
    install(TARGETS obs-parameter-animator RUNTIME DESTINATION obs-plugins/64bit LIBRARY DESTINATION obs-plugins/64bit)
    install(DIRECTORY data/ DESTINATION data/obs-plugins/obs-parameter-animator)
    if(OPA_FACE_TRACKING AND OpenCV_FOUND AND TARGET opencv_world)
      # Ship the matching release runtime next to the plugin DLL (get_target_property of the
      # import lib gives its .lib path; the DLL lives in the sibling bin folder).
      get_target_property(OPA_OCV_DLL opencv_world IMPORTED_LOCATION_RELEASE)
      if(OPA_OCV_DLL)
        get_filename_component(OPA_OCV_NAME "${OPA_OCV_DLL}" NAME_WE) # e.g. opencv_world4100
        get_filename_component(OPA_OCV_LIBDIR "${OPA_OCV_DLL}" DIRECTORY)
        get_filename_component(OPA_OCV_BINDIR "${OPA_OCV_LIBDIR}/../bin" ABSOLUTE)
        install(FILES "${OPA_OCV_BINDIR}/${OPA_OCV_NAME}.dll" DESTINATION obs-plugins/64bit OPTIONAL)
      endif()
    endif()
  elseif(UNIX AND NOT APPLE)
    include(GNUInstallDirs)
    install(TARGETS obs-parameter-animator LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/obs-plugins)
    install(DIRECTORY data/ DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/obs/obs-plugins/obs-parameter-animator)
  else()
    message(FATAL_ERROR "This package targets Windows x64 and Linux; macOS bundle/signing is not implemented")
  endif()
endif()
